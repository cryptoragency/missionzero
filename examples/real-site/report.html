const RATE_MAX = 30;
const _hits = new Map();
function _check(ip) {
  const now = Date.now();
  let a = (_hits.get(ip) || []).filter((t) => now - t < RATE_WINDOW_MS);
  a.push(now);
  _hits.set(ip, a);
  if (_hits.size > 2000) for (const [k, v] of _hits) if (!v.length || now - v[v.length - 1] > RATE_WINDOW_MS) _hits.delete(k);
  return a.length;
}
